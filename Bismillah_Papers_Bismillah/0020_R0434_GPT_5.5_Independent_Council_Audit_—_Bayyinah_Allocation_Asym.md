#!/usr/bin/env python3
import json, decimal, math
from collections import defaultdict, Counter
from pathlib import Path
D=decimal.Decimal
base=Path('/home/user/workspace/r6-gpt-5-5/repro')
rows=json.load(open(base/'snapshot/gpsc-qqsz_FY2022-2024.json'))
manifest=json.load(open(base/'snapshot/MANIFEST.json'))
fields=list(manifest['fields'])

def money(row, field):
    v=row.get(field)
    if v is None or v=='':
        return None
    return D(str(v))

def year(row):
    return row['fiscal_year_end_date'][:4]

def pct(n,d):
    return (D(n)/D(d)*D(100)) if d else D(0)

def gini(vals):
    xs=sorted(float(v) for v in vals if v>0)
    n=len(xs)
    total=sum(xs)
    if n==0 or total==0: return float('nan')
    weighted=sum((i+1)*x for i,x in enumerate(xs))
    return (2*weighted)/(n*total)-(n+1)/n

print('INDEPENDENT_STEP4_CODE_VERSION=2026-07-19b')
print('TOTAL_ROWS', len(rows))
fy2024=[r for r in rows if year(r)=='2024']
face_vals=[money(r,'contract_amount') for r in fy2024]
exp_vals=[money(r,'amount_expended_for_fiscal_year') for r in fy2024]
face_total=sum((v for v in face_vals if v is not None), D('0'))
exp_total=sum((v for v in exp_vals if v is not None), D('0'))
print('FY2024_ROW_COUNT', len(fy2024))
print('FY2024_FACE_TOTAL', f'{face_total:.2f}')
print('FY2024_ANNUAL_EXP_TOTAL', f'{exp_total:.2f}')
mw=[r for r in fy2024 if r.get('vendor_is_a_mwbe')=='Y']
mw_face=sum((money(r,'contract_amount') or D('0') for r in mw), D('0'))
mw_exp=sum((money(r,'amount_expended_for_fiscal_year') or D('0') for r in mw), D('0'))
print('FY2024_MWBE_COUNT', len(mw))
print('FY2024_MWBE_COUNT_SHARE_PERCENT', f'{pct(len(mw),len(fy2024)):.4f}')
print('FY2024_MWBE_FACE_SHARE_PERCENT', f'{pct(mw_face,face_total):.4f}')
print('FY2024_MWBE_ANNUAL_EXP_SHARE_PERCENT', f'{pct(mw_exp,exp_total):.4f}')
# vendor concentration and Gini: aggregate to one positive total per distinct nonblank vendor_name
for amount_field,label in [('contract_amount','FACE'),('amount_expended_for_fiscal_year','ANNUAL_EXP')]:
    totals=defaultdict(D)
    for r in fy2024:
        name=(r.get('vendor_name') or '').strip()
        if not name: continue
        val=money(r, amount_field)
        if val is None: continue
        totals[name]+=val
    positives=[v for v in totals.values() if v>0]
    if label=='FACE':
        top10=sum(sorted(positives, reverse=True)[:10], D('0'))
        print('FY2024_TOP10_VENDOR_FACE_CONCENTRATION_PERCENT', f'{pct(top10,face_total):.4f}')
    print(f'FY2024_{label}_POSITIVE_VENDOR_COUNT', len(positives))
    print(f'FY2024_{label}_VENDOR_GINI', f'{gini(positives):.5f}')
# Missing/zero for contract_amount in FY2024
miss=sum(1 for r in fy2024 if money(r,'contract_amount') is None)
zero=sum(1 for r in fy2024 if money(r,'contract_amount')==D('0'))
neg=sum(1 for r in fy2024 if (money(r,'contract_amount') is not None and money(r,'contract_amount')<0))
print('FY2024_CONTRACT_AMOUNT_MISSING_ZERO_NEGATIVE', miss, zero, neg)
exclude={'fiscal_year_end_date','amount_expended_for_fiscal_year','amount_expended_to_date','current_or_outstanding_balance'}
fingerprint_fields=[f for f in fields if f not in exclude]
print('FINGERPRINT_FIELD_COUNT', len(fingerprint_fields))
print('FINGERPRINT_FIELDS', '|'.join(fingerprint_fields))
sets18={}
setsall={}
for y in ['2022','2023','2024']:
    yr=[r for r in rows if year(r)==y]
    sets18[y]=set(tuple(r.get(f,'') for f in fingerprint_fields) for r in yr)
    setsall[y]=set(tuple(r.get(f,'') for f in fields) for r in yr)
for a,b in [('2022','2023'),('2022','2024'),('2023','2024')]:
    print(f'PAIRWISE_18FIELD_OVERLAP_FY{a[-2:]}_FY{b[-2:]}', len(sets18[a] & sets18[b]))
for a,b in [('2022','2023'),('2022','2024'),('2023','2024')]:
    print(f'PAIRWISE_ALLFIELD_OVERLAP_FY{a[-2:]}_FY{b[-2:]}', len(setsall[a] & setsall[b]))
print('ANY_PAIRWISE_ALLFIELD_OVERLAP_TOTAL', sum(len(setsall[a]&setsall[b]) for a,b in [('2022','2023'),('2022','2024'),('2023','2024')]))
