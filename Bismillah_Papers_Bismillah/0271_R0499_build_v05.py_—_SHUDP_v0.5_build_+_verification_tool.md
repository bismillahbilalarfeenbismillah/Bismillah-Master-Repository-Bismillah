def sha256(p):
    h = hashlib.sha256()
    with open(p, "rb") as f:
        for b in iter(lambda: f.read(65536), b""):
            h.update(b)
    return h.hexdigest()

def extract_yaml(src_text):
    blocks = re.findall(r"```yaml\n(.*?)\n```", src_text, re.DOTALL)
    if not blocks:
        raise ValueError("no yaml block found in source")
    return blocks[0]


def gate_yaml(src_text):
    import yaml
    y = extract_yaml(src_text)
    data = yaml.safe_load(y)          # real parser; raises on malformed YAML
    c = data["contract"]
    issues = []
    # four-axiom discharge paths
    for m in c["conservative_outcome_set"]:
        if "discharge" not in m:
            issues.append("outcome-set member without discharge: %s" % m.get("id"))
    # A5 block
    ei = c["evidence_independence"]
    req = ["acting_substrate", "safety_relevant_event_set_Estar",
           "completeness_boundary_B", "witness_set_W", "evidence_ledger_L",
           "withinauth_interpretation", "per_event_witness_commitment",
           "on_missing_independent_evidence_for_covered_event"]
    for k in req:
        if k not in ei:
            issues.append("A5 block missing member: %s" % k)
    if "temporal_floor" not in ei.get("completeness_boundary_B", {}):
        issues.append("A5 boundary B missing temporal_floor")
    if ei.get("withinauth_interpretation", {}).get("quantified_over") != "acting_substrate_S_only":
        issues.append("WithinAuth not quantified over acting substrate only")
    disc = ei.get("on_missing_independent_evidence_for_covered_event", {})
    if disc.get("axiom_level") != "A5_UNSATISFIED_nonconforming":
        issues.append("A5 missing-evidence axiom_level wrong")
    if disc.get("event_level") != "UNKNOWN":
        issues.append("A5 missing-evidence event_level not UNKNOWN")
    if ei.get("not_a_fifth_projection") is not True:
        issues.append("A5 not_a_fifth_projection flag not set")
    if ei.get("evidence_ledger_L", {}).get("reconstruction_from_subject_emitted_logs") != "forbidden":
        issues.append("A5 ledger does not forbid subject-emitted reconstruction")
    return issues

def md_inline_to_rl(text):
    """Minimal, safe inline markdown -> reportlab markup. Monochrome: links black+underline."""
    # escape XML
    text = text.replace("&", "&amp;").replace("<", "&lt;").replace(">", "&gt;")
    # protect literal 'E*' / 'E**' (the safety-relevant event set symbol) so the
    # asterisk is not read as markdown emphasis. Use a sentinel restored at the end.
    text = text.replace("E**", "\x00ESTAR\x00").replace("E*", "\x00ESTAR\x00")
    # bold **...**
    text = re.sub(r"\*\*(.+?)\*\*", r"<b>\1</b>", text)
    # italic *...* (single, not part of **)
    text = re.sub(r"(?<!\*)\*(?!\*)(.+?)(?<!\*)\*(?!\*)", r"<i>\1</i>", text)
    text = text.replace("\x00ESTAR\x00", "E*")
    # inline code `...`
    text = re.sub(r"`([^`]+?)`", r'<font face="Mono">\1</font>', text)
    # bare URLs -> clickable black link
    def linkify(m):
        url = m.group(0).rstrip(".,;)")
        trail = m.group(0)[len(url):]
        return '<a href="%s" color="black"><u>%s</u></a>%s' % (url, url, trail)
    text = re.sub(r"https?://[^\s)]+", linkify, text)
    return text

def gate_mono():
    """Zero chromatic color operators + zero chromatic pages."""
    import pypdfium2 as pdfium
    issues = []
    # raw color operators via qpdf QDF stream inspection
    raw = subprocess.run(["qpdf", "--qdf", "--object-streams=disable",
                          str(PDF), "-"], capture_output=True)
    content = raw.stdout
    # look for rg/RG (RGB) and k/K (CMYK) operators with non-gray values
    chromatic_ops = 0
    for m in re.finditer(rb"([\d.]+) ([\d.]+) ([\d.]+) (rg|RG)\b", content):
        r, g, b = float(m.group(1)), float(m.group(2)), float(m.group(3))
        if not (abs(r - g) < 1e-6 and abs(g - b) < 1e-6):
            chromatic_ops += 1
    for m in re.finditer(rb"([\d.]+) ([\d.]+) ([\d.]+) ([\d.]+) (k|K)\b", content):
        c, mm, y, kk = [float(m.group(x)) for x in range(1, 5)]
        if c > 1e-6 or mm > 1e-6 or y > 1e-6:
            chromatic_ops += 1
    if chromatic_ops:
        issues.append("chromatic color operators: %d" % chromatic_ops)
    # render pages, check for chromatic pixels
    pdf = pdfium.PdfDocument(str(PDF))
    chromatic_pages = []
    for pno in range(len(pdf)):
        page = pdf[pno]
        bmp = page.render(scale=1.0).to_numpy()  # HxWx{3,4}
        rgb = bmp[:, :, :3].astype("int16")
        rmax = rgb.max(axis=2); rmin = rgb.min(axis=2)
        sat = (rmax - rmin)
        if (sat > 16).sum() > 40:   # tolerance for antialiasing
            chromatic_pages.append(pno + 1)
    if chromatic_pages:
        issues.append("chromatic pages: %s" % chromatic_pages)
    return issues, len(pdf)


def gate_title():
    """Byte-exact full-title parity across source H1, PDF metadata, PDF cover text, HTML, DOCX."""
    from pypdf import PdfReader
    issues = []
    # canonical source first-line H1
    src = SRC.read_text().lstrip("\n")
    first = src.split("\n", 1)[0].strip()
    src_title = first[2:].strip() if first.startswith("# ") else ""
    if src_title != TITLE:
        issues.append("source H1 title != exact title (got %r)" % src_title)
    # PDF metadata
    md = PdfReader(str(PDF)).metadata
    if (md.get("/Title") or "") != TITLE:
        issues.append("PDF metadata /Title != exact title (got %r)" % md.get("/Title"))
    # PDF cover: full title text present on page 1 (whitespace-normalized), incl. the tail
    p1 = subprocess.run(["pdftotext", "-f", "1", "-l", "1", str(PDF), "-"],
                        capture_output=True, text=True).stdout
    p1n = re.sub(r"\s+", " ", p1)
    if re.sub(r"\s+", " ", TITLE) not in p1n:
        issues.append("PDF cover page does not contain the exact full title")
    if "Cross-Organizational Compromise" not in p1n:
        issues.append("PDF cover missing 'Cross-Organizational Compromise' tail")
    # HTML title element
    htext = HTML.read_text()
    m = re.search(r"<title>(.*?)</title>", htext, re.DOTALL)
    htitle = re.sub(r"\s+", " ", (m.group(1) if m else "")).strip()
    if htitle != re.sub(r"\s+", " ", TITLE):
        issues.append("HTML <title> != exact title (got %r)" % htitle)
    # DOCX core.xml title
    import zipfile
    z = zipfile.ZipFile(str(DOCX))
    try:
        core = z.read("docProps/core.xml").decode()
        mt = re.search(r"<dc:title>(.*?)</dc:title>", core, re.DOTALL)
        dtitle = re.sub(r"\s+", " ", (mt.group(1) if mt else "")).strip()
    except KeyError:
        dtitle = ""
    if dtitle != re.sub(r"\s+", " ", TITLE):
        issues.append("DOCX dc:title != exact title (got %r)" % dtitle)
    return issues


def gate_meta():
    from pypdf import PdfReader
    md = PdfReader(str(PDF)).metadata
    issues = []
    if (md.get("/Author") or "") != AUTHOR:
        issues.append("Author != %s (got %r)" % (AUTHOR, md.get("/Author")))
    if (md.get("/Title") or "") != TITLE:
        issues.append("Title mismatch (got %r)" % md.get("/Title"))
    subj = (md.get("/Subject") or "") + (md.get("/Keywords") or "")
    if "v0.5" not in subj:
        issues.append("version not in subject/keywords")
    return issues


def gate_concordance():
    """Section headings present across source, PDF text, HTML, DOCX text."""
    src = SRC.read_text()
    heads = re.findall(r"^#{1,3} (.+)$", src, re.MULTILINE)
    # normalize: strip markdown markers, take a distinctive token
    def norm(h):
        h = re.sub(r"\[v0\.5 post-freeze addition\]", "", h)
        h = re.sub(r"[*`_]", "", h)
        return h.strip()
    heads = [norm(h) for h in heads]
    ptext = pdf_text()
    htext = HTML.read_text()
    dtext = subprocess.run(["pandoc", str(DOCX), "-t", "plain"],
                           capture_output=True, text=True).stdout
    def flat(s):
        return re.sub(r"\s+", " ", s)
    pf, hf, df = flat(ptext), flat(htext), flat(dtext)
    issues = []
    for h in heads:
        key = flat(h)
        if len(key) < 4:
            continue
        miss = []
        if key not in pf: miss.append("PDF")
        if key not in hf: miss.append("HTML")
        if key not in df: miss.append("DOCX")
        if miss:
            issues.append("heading %r missing in %s" % (h, ",".join(miss)))
    return issues, len(heads)


def gate_content():
    issues = []
    ptext = pdf_text(); htext = HTML.read_text()
    dtext = subprocess.run(["pandoc", str(DOCX), "-t", "plain"],
                           capture_output=True, text=True).stdout
    for label, t in (("PDF", ptext), ("HTML", htext), ("DOCX", dtext)):
        nt = re.sub(r"\s+", " ", t)
        if FREEZE_PHRASE not in nt:
            issues.append("%s: evidence-freeze language missing" % label)
        # adjudication finality: A5 verdict must not be framed as a 'recommended verdict'
        low = nt.lower()
        if "recommended verdict" in low or "recommended formal verdict" in low:
            issues.append("%s: A5 verdict framed as a recommendation ('recommended verdict')" % label)
        if COMPANION_DOI not in re.sub(r"\s+", "", t):
            # DOI may wrap; check compact
            if COMPANION_DOI.replace("/", "") not in re.sub(r"[\s/]+", "", t):
                issues.append("%s: companion DOI missing" % label)
        # must NOT assert A5 is a fifth projection
        low = re.sub(r"\s+", " ", t).lower()
        if "fifth projection" in low:
            # allowed only in negations
            for m in re.finditer(r".{0,70}fifth projection.{0,20}", low):
                ctx = m.group(0)
                if not any(neg in ctx for neg in ["not a fifth projection", "not_a_fifth",
                                                   "not as a fifth", "not a fifth",
                                                   "not add a fifth", "no fifth", "not turn",
                                                   "does not add", "is not a fifth"]):
                    issues.append("%s: possible bad 'fifth projection' claim: ...%s..." % (label, ctx))
    return issues


def gate_dash():
    issues = []
    for p in [SRC]:
        t = p.read_text()
        for ch, name in (("\u2013", "en"), ("\u2014", "em"), ("\u2015", "horbar")):
            if ch in t:
                issues.append("%s: %s-dash present" % (p.name, name))
    return issues


def gate_byte():
    issues = []
    if sha256(BASELINE) != EXPECT_BASELINE_SHA:
        issues.append("baseline source not byte-identical")
    if sha256(AXSH5) != EXPECT_AXSH5_SHA:
        issues.append("AX-SH 5 PDF not byte-identical")
    return issues


def gate_qpdf():
    r = subprocess.run(["qpdf", "--check", str(PDF)], capture_output=True, text=True)
    ok = r.returncode == 0
    return ([] if ok else ["qpdf --check failed:\n" + r.stdout + r.stderr]), r.stdout


def run_verify():
    print("\n===== SHUDP v0.5 GATE SUITE =====")
    results = {}
    src = SRC.read_text()
    y = gate_yaml(src);       results["G-YAML"] = y
    d = gate_dash();          results["G-DASH"] = d
    b = gate_byte();          results["G-BYTE"] = b
    if PDF.exists():
        m = gate_meta();          results["G-META"] = m
        mono, npages = gate_mono();  results["G-MONO"] = mono
        q, qtxt = gate_qpdf();    results["G-QPDF"] = q
    else:
        print("PDF missing; skipping PDF gates")
    if PDF.exists() and HTML.exists() and DOCX.exists():
        results["G-TITLE"] = gate_title()
        conc, nh = gate_concordance(); results["G-CONCORDANCE"] = conc
        cont = gate_content();    results["G-CONTENT(freeze/DOI/nofifth/final)"] = cont
    allpass = True
    for k, v in results.items():
        status = "PASS" if not v else "FAIL"
        if v: allpass = False
        print("  %-32s %s" % (k, status))
        for it in v:
            print("       - %s" % it)
    print("=================================")
    print("OVERALL:", "PASS" if allpass else "FAIL")
    return allpass
