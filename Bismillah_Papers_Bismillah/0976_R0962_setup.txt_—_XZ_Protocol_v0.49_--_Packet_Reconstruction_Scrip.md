#!/usr/bin/env python3
"""
XZ Protocol v0.49 -- Packet Reconstruction Script
Reconstructs the full 17-member packet from the 10-file .txt dispatch format.
Run this FIRST before build.py or any test suite.

Usage:
    python3 setup.txt

This script:
  1. Splits code_modules.txt into 6 individual .py files
  2. Splits test_suites.txt into 6 individual .py files
  3. Renames singleton .txt files to their canonical names
  4. Verifies all 15 file digests against PACKET_SHA256SUMS
  5. Reports any mismatches
"""

import hashlib
import os
import re
import shutil
import sys

DISPATCH_DIR = os.path.dirname(os.path.abspath(__file__))
os.chdir(DISPATCH_DIR)

# ---------- Step 0: Rename singletons ----------

RENAMES = {
    "protocol.txt": "xz_protocol_v0_49.md",
    "build.txt": "build.py",
    "manifest.txt": "PACKET_SHA256SUMS",
    "history.txt": "HISTORY.sha256",
    "trust_roots.txt": "xz_protocol_trust_roots_PUBLIC.json",
}

print("=" * 70)
print("XZ Protocol v0.49 -- Packet Reconstruction")
print("=" * 70)
print()

for src, dst in RENAMES.items():
    if os.path.exists(src):
        shutil.copy2(src, dst)
        print(f"  [OK] {src} -> {dst}")
    elif os.path.exists(dst):
        print(f"  [OK] {dst} already exists")
    else:
        print(f"  [MISS] {src} not found", file=sys.stderr)

print()

# ---------- Step 1: Split bundles ----------

MARKER_BEGIN = re.compile(
    r"^===== BEGIN FILE: (\S+) \| SIZE: (\d+) bytes \| SHA256: ([0-9a-f]{64}) =====$"
)
MARKER_END = re.compile(r"^===== END FILE: (\S+) =====$")


def split_bundle(bundle_path, label):
    """Split a bundled .txt file into individual files."""
    if not os.path.exists(bundle_path):
        print(f"  [MISS] {bundle_path} not found", file=sys.stderr)
        return []

    with open(bundle_path, "r", encoding="utf-8") as f:
        lines = f.readlines()

    files_written = []
    current_file = None
    current_lines = []
    expected_digest = None

    for line in lines:
        m_begin = MARKER_BEGIN.match(line.rstrip("\n"))
        m_end = MARKER_END.match(line.rstrip("\n"))

        if m_begin:
            current_file = m_begin.group(1)
            expected_digest = m_begin.group(3)
            current_lines = []
        elif m_end and current_file:
            # Remove trailing empty line if the marker added one
            content = "".join(current_lines)
            if content.endswith("\n\n"):
                # The bundle adds a newline after the file; strip the extra
                pass  # keep as-is, the original file may or may not end with \n
            # Write the file
            # Strip exactly one trailing newline that was added by the bundler
            # if the original didn't have a double newline
            if content.endswith("\n"):
                content_bytes = content.encode("utf-8")
            else:
                content_bytes = content.encode("utf-8")

            # Verify against expected digest
            actual_digest = hashlib.sha256(content_bytes).hexdigest()
            if actual_digest != expected_digest:
                # Try stripping the last newline (bundler may have added one)
                alt_bytes = content.rstrip("\n").encode("utf-8") + b"\n"
                alt_digest = hashlib.sha256(alt_bytes).hexdigest()
                if alt_digest == expected_digest:
                    content_bytes = alt_bytes
                    actual_digest = alt_digest

            with open(current_file, "wb") as out:
                out.write(content_bytes)

            status = "OK" if actual_digest == expected_digest else "DIGEST MISMATCH"
            print(f"  [{status}] {current_file} ({len(content_bytes)} bytes)")
            if status != "OK":
                print(f"           expected: {expected_digest}")
                print(f"           actual:   {actual_digest}")

            files_written.append(current_file)
            current_file = None
            current_lines = []
        elif current_file is not None:
            current_lines.append(line)

    return files_written


print(f"Splitting code_modules.txt:")
code_files = split_bundle("code_modules.txt", "code")
print(f"  -> {len(code_files)} code modules\n")

print(f"Splitting test_suites.txt:")
test_files = split_bundle("test_suites.txt", "test")
print(f"  -> {len(test_files)} test suites\n")

# ---------- Step 2: Verify against PACKET_SHA256SUMS ----------

print("Verifying against PACKET_SHA256SUMS:")
manifest = "PACKET_SHA256SUMS"
if not os.path.exists(manifest):
    print(f"  [MISS] {manifest} not found -- cannot verify", file=sys.stderr)
    sys.exit(1)

ok_count = 0
fail_count = 0
missing_count = 0

with open(manifest) as f:
    for line in f:
        line = line.strip()
        if not line:
            continue
        parts = line.split("  ", 1)
        if len(parts) != 2:
            continue
        expected_hash, filename = parts
        if not os.path.exists(filename):
            print(f"  [MISS] {filename}")
            missing_count += 1
            continue
        actual_hash = hashlib.sha256(open(filename, "rb").read()).hexdigest()
        if actual_hash == expected_hash:
            print(f"  [OK] {filename}")
            ok_count += 1
        else:
            print(f"  [FAIL] {filename}")
            print(f"         expected: {expected_hash}")
            print(f"         actual:   {actual_hash}")
            fail_count += 1

print()
manifest_digest = hashlib.sha256(open(manifest, "rb").read()).hexdigest()
print(f"PACKET_SHA256SUMS digest: {manifest_digest}")
print()
print(f"Result: {ok_count} OK / {fail_count} FAIL / {missing_count} MISSING")

if fail_count == 0 and missing_count == 0:
    print("\n*** RECONSTRUCTION VERIFIED -- all 15 files match ***")
    print("\nNext steps:")
    print("  python3 build.py          # run build verification (194 checks)")
    print("  python3 -m pytest -q      # run 5 standard test suites (117 tests)")
    print("  python3 -m pytest xz_protocol_phase_a_tests.py -q  # phase A (69 tests)")
    print("  # Total: 186 tests across 6 suites")
else:
    print("\n*** RECONSTRUCTION FAILED -- do not proceed ***")
    sys.exit(1)
