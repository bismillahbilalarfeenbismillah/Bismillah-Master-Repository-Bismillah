def sha256_file(path):
    h = hashlib.sha256()
    with open(path, "rb") as f:
        for chunk in iter(lambda: f.read(65536), b""):
            h.update(chunk)
    return h.hexdigest()

def sha256_string(s):
    return hashlib.sha256(s.encode()).hexdigest()

def get_body(protocol):
    marker = "## 27."
    positions = [m.start() for m in re.finditer(re.escape(marker), protocol)]
    if not positions:
        errors.append("FAIL: Section 27 heading not found")
        return protocol
    heading_positions = [p for p in positions if p == 0 or protocol[p-1] == '\n']
    if len(heading_positions) != 1:
        errors.append(f"FAIL: Expected 1 line-starting '## 27.' heading, found {len(heading_positions)}")
        return protocol
    body = protocol[:heading_positions[0]]
    if not body or len(body) >= len(protocol):
        errors.append("FAIL: Body is empty or equals entire protocol")
    return body

def check(condition, msg):

def _strip_entry(s27, ver):
    """Remove entry v0.<ver> from Section 27 text."""
    m = re.search(r'(\n- \*\*v0\.' + str(ver) + r'[^*]*\*\*.*?)(?=\n- \*\*v0\.|\n\*Provenance|\Z)',
                  s27, re.DOTALL)
    if not m:
        return s27
    return s27[:m.start()] + s27[m.end():]

def _full_seal(s27):
    """Compute FULL_SECTION_27 hash."""
    prov = s27.find("*Provenance and review status.*")
    full = s27[:prov].strip() if prov > 0 else s27.strip()
    return hashlib.sha256(full.encode('utf-8')).hexdigest()

def _entry_text(s27, ver):
    """Extract entry v0.<ver> text."""
    m = re.search(r'(- \*\*v0\.' + str(ver) + r'[^*]*\*\*.*?)(?=\n- \*\*v0\.|\n\*Provenance|\Z)',
                  s27, re.DOTALL)
    return m.group(1) if m else None

def derive_prior_history(protocol, history_lines):
    """F-OP21S-01 fix: recursive PRIOR_HISTORY derivation.

    Reconstructs the prior version's HISTORY.sha256 by chain recursion.
    At each level: strip entries above target_ver, compute FULL_SECTION_27,
    assemble seals, and find the PRIOR_HISTORY value either by scraping
    the target entry's prose or by recursing one level deeper.

    Terminates at v0.18 (no PRIOR_HISTORY line) or at any entry whose prose
    contains a scrapeable PRIOR_HISTORY + 64-hex value.
    """
    s27 = protocol[protocol.find("## 27."):]
    current_ver = int(VERSION.replace("0.", ""))
    # Collect all entry seals from HISTORY.sha256
    entry_seals = {}
    for line in history_lines:
        if '  ENTRY_v0.' in line and not line.startswith("PRIOR_HISTORY"):
            parts = line.split('  ', 1)
            if len(parts) == 2:
                entry_seals[parts[1]] = parts[0]

    def reconstruct(target_ver):
        """Return sha256 of what HISTORY.sha256 would have been at target_ver."""
        # Strip all entries above target_ver
        working = s27
        for v in range(current_ver, target_ver, -1):
            working = _strip_entry(working, v)
        full = _full_seal(working)
        lines = []
        for i in range(1, target_ver + 1):
            label = f"ENTRY_v0.{i}"
            if label in entry_seals:
                lines.append(f"{entry_seals[label]}  {label}")
        lines.append(f"{full}  FULL_SECTION_27")
        # Find the PRIOR_HISTORY value for this version
        if target_ver >= 19:
            # Try scraping from the target entry's prose
            et = _entry_text(s27, target_ver)
            if et:
                ph = re.search(r'PRIOR_HISTORY[^\n]*?([0-9a-f]{64})', et)
                if not ph:
                    # Also try parenthetical format: (41428758...)
                    ph = re.search(r'HISTORY\.sha256\s+(?:file\s+)?\(([0-9a-f]{64})', et)
                if ph:
                    lines.append(f"PRIOR_HISTORY {ph.group(1)}")
                elif target_ver > 19:
                    # Recurse: derive the prior-prior hash
                    prior_hash = reconstruct(target_ver - 1)
                    if prior_hash:
                        lines.append(f"PRIOR_HISTORY {prior_hash}")
                # else: v0.19 with no scrapeable value -- try the cb67f222 from its entry
                else:
                    cb = re.search(r'cb67f222[0-9a-f]{56}', et)
                    if cb:
                        lines.append(f"PRIOR_HISTORY {cb.group(0)}")
        # v0.18 and earlier: no PRIOR_HISTORY line
        content = "\n".join(lines) + "\n"
        return hashlib.sha256(content.encode('utf-8')).hexdigest()

    return reconstruct(current_ver - 1)
